		private Shape BuildRectanglePath(RectangleGeometry area)
		{
			var result = new Path()
			{
				Fill = new SolidColorBrush(Colors.Transparent),
				Stroke = Brushes.Transparent,
				StrokeThickness = 0,
				Data = area,
				IsHitTestVisible = true,
				Opacity = 1.0
			};

			return result;
		}

		private Rect BuildRect(Rect r, SizeDbl contentScale)
		{
			var result = new Rect(r.Location, r.Size);
			result.Scale(contentScale.Width, contentScale.Height);

			return result;
		}


CbBlendedColorPair

		private Shape BuildRectanglePath(RectangleGeometry area, ColorBandColor startColor, ColorBandColor endColor, bool blend)
		{
			var result = new Path()
			{
				Fill = GetBlendedBrush(startColor, endColor, blend),
				Stroke = Brushes.Transparent,
				StrokeThickness = 0,
				Data = area,
				IsHitTestVisible = true,
				Opacity = 1.0
			};

			return result;
		}

	Both the BlendedBandRectangle and _containerPath
		a. Have their own Path created
		b. from their own RectangleGeometry
		c. Using the same, shared Rect
		d. Set the ZIndex using the BLENDED_AREA_ZINDEX_DELTA

	The CBRectangle's Path is Transparent, Hit-Testable, and Opaque

	The CbBlendedColorPair's Path is Gradient Fill, Hit-Testable and Opaque.


	When the Opacity is set, both receive the same value.

	When the ZIndex is set, both receive the same value == BaseVal + BLENDED_AREA_ZINDEX_DELTA
